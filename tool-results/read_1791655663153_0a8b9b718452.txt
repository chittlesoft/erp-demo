    40→    40→  private isAncestor(candidate: number, target: number): boolean {

    43→    43→      if (l.componentItemId === target || this.isAncestor(l.componentItemId, target)) return true;

    70→    70→    if (!bal || bal.qty < qty - 1e-9) throw new ErpError("Insufficient stock in source location for this batch");

    73→    73→      this.addTxn({ date: today(), itemId, warehouseId: fromWh, qty: -qty, txnType: "TRANSFER", ref, batchNo, unitCost: cost });

    87→    87→  orderRequirements(order: ProductionOrder): Requirement[] {
    88→    88→    const stock = this.onHand();
    89→    89→    return this.listBom(order.itemId).map(l => {
    90→    90→      const it = this.getItem(l.componentItemId)!;
    91→    91→      const required = roundQty(l.qty * (1 + l.scrapPct / 100) * order.qty, it.uom);
    92→    92→      const available = stock.get(it.id) ?? 0;
    93→    93→      return { itemId: it.id, code: it.code, name: it.name, uom: it.uom, required, available, shortage: Math.max(0, +(required - available).toFixed(3)) };
    94→    94→    });

    96→    96→  startOrder(id: number): ProductionOrder {
    97→    97→    const order = this.getOrder(id);
    98→    98→    if (!order) throw new ErpError("Order not found");
    99→    99→    if (order.status !== "PLANNED") throw new ErpError("Only planned orders can be started");
   100→   100→    const reqs = this.orderRequirements(order);
   101→   101→    const short = reqs.filter(r => r.shortage > 0);
   102→   102→    if (short.length) throw new ErpError(`Material shortage: ${short.map(s => `${s.code} (${s.shortage} ${s.uom})`).join(", ")}`);
   103→   103→    const blocked = new Set(this.listWarehouses().filter(w => ["QC", "REJ", "JW"].includes(w.code)).map(w => w.id));
   104→   104→    const consumed: { itemId: number; batchNo: string | null; qty: number }[] = [];
   105→   105→    sqlite.transaction(() => {
   106→   106→      const batches = this.stockByBatch().filter(b => !blocked.has(b.warehouseId));
   107→   107→      for (const r of reqs) {
   108→   108→        let need = r.required;
   109→   109→        const cost = this.getItem(r.itemId)?.stdCost ?? 0;
   110→   110→        for (const b of batches.filter(b => b.itemId === r.itemId)) { // FIFO
   111→   111→          if (need <= 1e-9) break;
   112→   112→          const take = Math.min(need, b.qty);
   113→   113→          this.addTxn({ date: today(), itemId: r.itemId, warehouseId: b.warehouseId, qty: -take, txnType: "ISSUE", ref: order.orderNo, batchNo: b.batchNo, unitCost: cost });
   114→   114→          consumed.push({ itemId: r.itemId, batchNo: b.batchNo, qty: +take.toFixed(3) });
   115→   115→          need -= take;
   116→   116→        }
   117→   117→      }
   118→   118→      db.update(productionOrders).set({ status: "IN_PROGRESS", consumedBatches: JSON.stringify(consumed) }).where(eq(productionOrders.id, id)).run();
   119→   119→    })();
   120→   120→    return this.getOrder(id)!;
   121→   121→  }
   122→   122→  completeOrder(id: number, goodQty: number, rejectQty: number): ProductionOrder {
   123→   123→    const order = this.getOrder(id);
   124→   124→    if (!order) throw new ErpError("Order not found");
   125→   125→    if (order.status !== "IN_PROGRESS") throw new ErpError("Only in-progress orders can be completed");
   126→   126→    const hasWip = !!sqlite.prepare(`SELECT name FROM sqlite_master WHERE type='table' AND name='wip_units'`).get();
   127→   127→    if (hasWip && (sqlite.prepare(`SELECT COUNT(*) c FROM wip_units WHERE order_id = ?`).get(id) as any).c > 0) throw new ErpError("This order is on the shop floor — close it from Shop Floor once all units are finished");
   128→   128→    if (goodQty + rejectQty !== order.qty) throw new ErpError(`Good + rejected must equal order quantity (${order.qty})`);
   129→   129→    const item = this.getItem(order.itemId)!;
   130→   130→    sqlite.transaction(() => {
   131→   131→      if (goodQty > 0) this.addTxn({ date: today(), itemId: item.id, warehouseId: order.warehouseId, qty: goodQty, txnType: "PROD_OUT", ref: order.orderNo, batchNo: order.orderNo, unitCost: this.rolledCost(item.id) });
   132→   132→      if (item.tracking === "serial") this.generateSerials(order, item, goodQty, rejectQty);
   133→   133→      db.update(productionOrders).set({ status: "COMPLETED", goodQty, rejectQty }).where(eq(productionOrders.id, id)).run();
   134→   134→    })();
   135→   135→    return this.getOrder(id)!;
   136→   136→  }
   137→   137→  private generateSerials(order: ProductionOrder, item: Item, good: number, bad: number, date = today()) {
   138→   138→    const ym = date.slice(2, 4) + date.slice(5, 7);
   139→   139→    const prefix = `${PREFIX[item.productLine ?? ""] ?? "FG"}-${ym}-`;
   140→   140→    const hasWip = !!sqlite.prepare(`SELECT name FROM sqlite_master WHERE type='table' AND name='wip_units'`).get();
   141→   141→    let seq = (sqlite.prepare(`SELECT MAX(n) m FROM (SELECT CAST(substr(serial_no, ?) AS INTEGER) n FROM serials WHERE serial_no LIKE ?${hasWip ? " UNION ALL SELECT CAST(substr(serial_no, ?) AS INTEGER) n FROM wip_units WHERE serial_no LIKE ?" : ""})`)
   142→   142→      .get(...(hasWip ? [prefix.length + 1, prefix + "%", prefix.length + 1, prefix + "%"] : [prefix.length + 1, prefix + "%"])) as any).m ?? 0;
   143→   143→    const results = [...Array(good).fill("PASS"), ...Array(bad).fill("FAIL")];
   144→   144→    for (const res of results) {
   145→   145→      seq++;
   146→   146→      db.insert(serials).values({
   147→   147→        serialNo: prefix + String(seq).padStart(5, "0"), itemId: item.id, productionOrderId: order.id,
   148→   148→        eolResult: res, eolData: JSON.stringify(simulatedEol(item.productLine, res === "PASS")), date, status: res === "PASS" ? "IN_STOCK" : "REJECTED",
   149→   149→      }).run();
   150→   150→    }
   151→   151→  }
   152→   152→  listSerials(limit = 300): Serial[] { return db.select().from(serials).orderBy(desc(serials.id)).limit(limit).all(); }
   153→   153→  traceSerial(serialNo: string) {
   154→   154→    const s = db.select().from(serials).where(eq(serials.serialNo, serialNo.trim().toUpperCase())).get();
   155→   155→    if (!s) return undefined;
   156→   156→    const order = this.getOrder(s.productionOrderId)!;
   157→   157→    const comps = (JSON.parse(order.consumedBatches) as { itemId: number; batchNo: string | null; qty: number }[])
   158→   158→      .map(c => ({ ...c, item: this.getItem(c.itemId) }));
   159→   159→    const inv = s.invoiceId ? (sqlite.prepare(`SELECT i.inv_no invNo, i.date, p.name customer FROM invoices i JOIN parties p ON p.id = i.customer_id WHERE i.id = ?`).get(s.invoiceId) as any) : null;
   160→   160→    return { serial: s, item: this.getItem(s.itemId), order, components: comps, dispatch: inv };
   161→   161→  }
   162→   162→  traceBatch(batchNo: string) {
   163→   163→    // reverse trace: which orders/serials consumed this raw-material batch
   164→   164→    const orders = this.listOrders().filter(o => (JSON.parse(o.consumedBatches) as any[]).some(c => c.batchNo === batchNo.trim()));
   165→   165→    return orders.map(o => ({ order: o, item: this.getItem(o.itemId), serials: db.select().from(serials).where(eq(serials.productionOrderId, o.id)).all() }));

   168→   168→  // ---------- MRP (multi-level, netted against stock) ----------
   169→   169→  mrp(): Requirement[] {
   170→   170→    const stock = this.onHand();
   171→   171→    const free = new Map(stock);
   172→   172→    const gross = new Map<number, number>();
   173→   173→    const explode = (itemId: number, qty: number) => {
   174→   174→      for (const l of this.listBom(itemId)) {
   175→   175→        const need = roundQty(l.qty * (1 + l.scrapPct / 100) * qty, this.getItem(l.componentItemId)!.uom);
   176→   176→        gross.set(l.componentItemId, (gross.get(l.componentItemId) ?? 0) + need);
   177→   177→        const avail = free.get(l.componentItemId) ?? 0;
   178→   178→        const used = Math.min(avail, need);
   179→   179→        free.set(l.componentItemId, avail - used);
   180→   180→        const net = need - used;
   181→   181→        if (net > 0 && this.listBom(l.componentItemId).length) explode(l.componentItemId, net);
   182→   182→      }
   183→   183→    };
   184→   184→    for (const o of this.listOrders().filter(o => o.status === "PLANNED")) explode(o.itemId, o.qty);
   185→   185→    return Array.from(gross.entries()).map(([itemId, required]) => {
   186→   186→      const it = this.getItem(itemId)!;
   187→   187→      const available = stock.get(itemId) ?? 0;
   188→   188→      return { itemId, code: it.code, name: it.name, uom: it.uom, required: +required.toFixed(3), available: +available.toFixed(3), shortage: Math.max(0, +(required - available).toFixed(3)) };
   189→   189→    }).sort((a, b) => b.shortage - a.shortage);
   190→   190→  }

   193→   193→  dashboard() {
   194→   194→    const all = this.listItems();
   195→   195→    const stock = this.onHand();
   196→   196→    const stockValue = all.reduce((s, i) => s + (stock.get(i.id) ?? 0) * (i.type === "FG" || i.type === "SFG" ? this.rolledCost(i.id) : i.stdCost), 0);
   197→   197→    const lowStock = all.filter(i => i.reorderLevel > 0 && (stock.get(i.id) ?? 0) < i.reorderLevel)
   198→   198→      .map(i => ({ id: i.id, code: i.code, name: i.name, uom: i.uom, onHand: stock.get(i.id) ?? 0, reorderLevel: i.reorderLevel }));
   199→   199→    const orders = this.listOrders();
   200→   200→    const done = orders.filter(o => o.status === "COMPLETED");
   201→   201→    const good = done.reduce((s, o) => s + o.goodQty, 0);
   202→   202→    const rej = done.reduce((s, o) => s + o.rejectQty, 0);
   203→   203→    const byLine: Record<string, { line: string; good: number; reject: number }> = {};
   204→   204→    for (const o of done) {
   205→   205→      const line = this.getItem(o.itemId)?.productLine ?? "Other";
   206→   206→      byLine[line] ??= { line, good: 0, reject: 0 };
   207→   207→      byLine[line].good += o.goodQty; byLine[line].reject += o.rejectQty;
   208→   208→    }
   209→   209→    return {
   210→   210→      itemCount: all.length, stockValue: Math.round(stockValue), lowStock,
   211→   211→      openOrders: orders.filter(o => o.status !== "COMPLETED").length,
   212→   212→      inProgress: orders.filter(o => o.status === "IN_PROGRESS").length,
   213→   213→      unitsGood: good, unitsRejected: rej, fpy: good + rej ? +(100 * good / (good + rej)).toFixed(1) : null,
   214→   214→      byLine: Object.values(byLine),
   215→   215→    };
   216→   216→  }
